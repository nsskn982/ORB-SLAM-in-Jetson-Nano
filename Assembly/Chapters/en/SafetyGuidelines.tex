\chapter{Safety Guidelines}

\section{Assembly Safety}

\begin{itemize}
    \item Disconnect battery before making electrical connections
    \item Wear safety goggles when working with power tools
    \item Do not test motors with fingers near moving parts
    \item Use anti-static wrist strap when handling Jetson Nano
\end{itemize}

\section{Operational Safety}

\begin{itemize}
    \item Operate robot only in clear, obstacle-free areas during testing
    \item Never reach into or block the robot while motors are powered
    \item Keep long hair and loose clothing away from spinning wheels
    \item Maintain full visibility of robot trajectory during autonomous operation
\end{itemize}

\section{Electrical Safety}

\begin{itemize}
    \item Use approved power supply only; do not attempt to modify voltage
    \item Check for exposed wires; insulate with tape if necessary
    \item Do not operate in wet environments
    \item Allow system to cool if running continuously longer than 1 hour
\end{itemize}

\section{Battery Handling}

\begin{itemize}
    \item Do not short-circuit battery terminals
    \item Do not dispose of battery in regular trash; use proper recycling
    \item Stop use immediately if battery shows signs of damage, swelling, or overheating
\end{itemize}
