\chapter{Maintenance}

\section{Routine Hardware Maintenance}

\subsection{Camera Lens Care}
\begin{itemize}
    \item Clean lens weekly with lens cloth and appropriate cleaning solution
    \item Check focus clarity by capturing test images
    \item Replace lens if scratches affect image quality
\end{itemize}

\subsection{Motor and Encoder Service}
\begin{itemize}
    \item Inspect wheelbase for debris or buildup (monthly)
    \item Test motor smoothness and listen for grinding sounds
    \item Check encoder pulse consistency with test run
    \item Replace motor brushes if performance degrades
\end{itemize}

\subsection{Structural Inspection}
\begin{itemize}
    \item Check all fasteners for looseness after high-acceleration runs
    \item Inspect caster ball for wear
    \item Verify camera mount rigidity
\end{itemize}

\section{Battery Management}

\begin{itemize}
    \item Recharge after each use session
    \item Check for swelling or damage
    \item Replace if capacity drops below 80\%
\end{itemize}
