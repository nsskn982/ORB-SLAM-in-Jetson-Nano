\chapter{Set Up}

\section{System Initialization}

\begin{enumerate}
    \item Power on the robot and wait for Jetson Nano to boot (30-60 seconds)
    \item Connect to robot via SSH: \texttt{ssh jetson@<robot\_ip>}
    \item Launch localization software: \texttt{roslaunch localization system.launch}
    \item Open web dashboard: \texttt{http://<robot\_ip>:8080}
\end{enumerate}

\section{Environment Preparation}

\begin{itemize}
    \item Place robot in test arena with known dimensions
    \item Ensure adequate lighting (minimum 500 lux recommended)
    \item Add visual landmarks or textured features
    \item Clear obstacles if in autonomous mode
\end{itemize}

\section{Calibration Procedure}

\begin{enumerate}
    \item Run camera calibration checkerboard scan
    \item Perform encoder zero calibration
    \item Execute test path to verify baseline accuracy
    \item Adjust feature extraction parameters if needed
\end{enumerate}
