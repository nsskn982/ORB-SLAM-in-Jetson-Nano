\section{Operating Modes}

\begin{itemize}
    \item \textbf{Visual Odometry}: Estimates relative pose at 30 Hz
    \item \textbf{Continuous Mapping}: Builds feature map during operation
    \item \textbf{Localization Against Map}: Position-only estimation when map exists
    \item \textbf{Loop Closure Detection}: Triggered when environment revisited
\end{itemize}

\section{Output Data Format}

\begin{itemize}
    \item Estimated Position: (x, y, theta) in meters and radians
    \item Confidence: 0.0--1.0 (1.0 = fully confident)
    \item Covariance: 3x3 pose uncertainty matrix
    \item Timestamp: Synchronized with camera frame capture
\end{itemize}
