\chapter{Add Ons}

\section{Enhanced Localization Features}

\subsection{Multi-Sensor Fusion Mode}
\begin{itemize}
    \item When IMU available: Uses gyro for drift-free heading
    \item When encoders available: Fuses with visual odometry for robustness
    \item When both available: Achieves best-in-class accuracy
    \item Enable via config: \texttt{sensor\_fusion: true}
\end{itemize}

\subsection{3D Mapping and Height Awareness}
\begin{itemize}
    \item Stereo camera provides depth for each visual feature
    \item Enables stair and ramp detection
    \item Improves obstacle avoidance in complex 3D environments
    \item Requires \texttt{mapping\_3d: true} in config
\end{itemize}

\subsection{Global Localization (Lidar-based)}
\begin{itemize}
    \item Use Lidar scans to match against precomputed map occupancy grids
    \item Enables position recovery if visual tracking is lost
    \item Can relocalize robot anywhere in mapped environment
    \item Access via web dashboard: click \"Global Relocalize\" button
\end{itemize}

\section{Performance Tuning Options}

\begin{itemize}
    \item Reduced feature extraction rate for low-power mode (15 Hz instead of 30 Hz)
    \item Aggressive loop closure for tightening accuracy \texttt{loop\_closure\_aggressiveness: high}
    \item Conservative mode for safety (requires strong evidence before transitions)
\end{itemize}
