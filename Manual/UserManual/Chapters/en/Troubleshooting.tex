\chapter{Troubleshooting}

\section{Localization Accuracy Issues}

\begin{itemize}
    \item \textbf{High position drift}: Increase feature extraction quality or optimize camera calibration
    \item \textbf{Loop closure fails}: Environment lacks distinctive features; add markers or increase detection sensitivity
    \item \textbf{Encoder drift dominates}: Visual features may be unreliable; check lighting or add reference landmarks
\end{itemize}

\section{Performance Problems}

\begin{itemize}
    \item \textbf{Slow frame processing}: Too many features extracted; adjust threshold
    \item \textbf{High CPU usage}: Disable feature visualization or reduce resolution
    \item \textbf{Memory exhaustion}: Prune old feature maps from memory; consider streaming to disk
\end{itemize}

\section{Sensor Failures}

\begin{itemize}
    \item \textbf{Encoder dropout}: Check wheel spin under no load; verify encoder connections
    \item \textbf{Feature extraction fails}: Inadequate lighting; switch to robust feature detector (ORB vs SIFT)
    \item \textbf{System crashes}: Log dump file location provided in error message; report to development team
\end{itemize}

\section{Recovery Steps}

\begin{enumerate}
    \item Restart localization system: \texttt{rosnode kill localization\_node}
    \item Clear cached data: \texttt{rm -rf ~/.ros/localization\_cache}
    \item Re-run calibration procedure
    \item If persistent, reflash SD card image
\end{enumerate}
