\chapter{Troubleshooting}

\section{Debugging Tools and Techniques}

\subsection{Logging and Visualization}
\begin{itemize}
    \item Enable debug logging: \texttt{export ROS\_LOG\_LEVEL=debug}
    \item Visualize pose graph: \texttt{rosrun g2o viewer}
    \item Plot trajectories with Python scripts in \texttt{scripts/}
    \item Use \texttt{rqt\_graph} to verify message flow
\end{itemize}

\subsection{Common Algorithmic Issues}
\begin{itemize}
    \item \textbf{Features not detected}: Check image histogram; may need exposure adjustment
    \item \textbf{Matches appear random}: Feature descriptor mismatch; verify matcher threshold
    \item \textbf{Pose estimates explode}: Check for numerical instability in transformation estimation
    \item \textbf{Bundle adjustment diverges}: Initial guess too far from solution; relax convergence tolerance initially
\end{itemize}

\section{Performance Debugging}

\begin{itemize}
    \item Profile with \texttt{perf} and \texttt{callgrind} to identify bottlenecks
    \item Use \texttt{top} to monitor Jetson CPU/GPU/memory during runs
    \item Check TensorRT optimization if using GPU-accelerated feature extraction
    \item Consider OpenCV CUDA support for real-time performance
\end{itemize}

\section{Test Regression Detection}

\begin{itemize}
    \item Maintain benchmark trajectories in \texttt{test/data/}
    \item Run \texttt{./test/run\_benchmarks.sh} after major changes
    \item Compare Absolute Trajectory Error against baseline
    \item Document acceptable regression thresholds
\end{itemize}
