\chapter{Maintenance}

\section{Code Repository Maintenance}

\subsection{Branch Strategy}
\begin{itemize}
    \item \texttt{master}: Stable release code, tested and documented
    \item \texttt{develop}: Integration branch for completed features
    \item \texttt{feature/*}: Individual feature development
\end{itemize}

\subsection{Contribution Workflow}
\begin{enumerate}
    \item Create feature branch from develop
    \item Implement and test locally
    \item Submit pull request with test results
    \item Address code review comments
    \item Merge to develop after approval
\end{enumerate}
\begin{figure}[H]
	\centering
	\begin{tikzpicture}[
		stepbox/.style={rectangle, draw=black!60, fill=blue!5, thick, text width=3.5cm, text centered, rounded corners, minimum height=1.2cm},

		]
		\node[stepbox] (s1) {\textbf{1. Create Branch}\\From \texttt{develop}};
		\node[stepbox, right=1cm of s1] (s2) {\textbf{2. Implement}\\Write and test locally};
		\node[stepbox, right=1cm of s2] (s3) {\textbf{3. Pull Request}\\Submit test results};
		\node[stepbox, below=1cm of s3] (s4) {\textbf{4. Code Review}\\Address comments};
		\node[stepbox, left=1cm of s4, fill=green!10, draw=green!60!black] (s5) {\textbf{5. Merge}\\Merge to \texttt{develop}};
		
		\draw[arrow] (s1) -- (s2);
		\draw[arrow] (s2) -- (s3);
		\draw[arrow] (s3) -- (s4);

		\draw[arrow, dashed] (s4.south) -- ++(0,-0.5) -| node[pos=0.25, below, font=\small] {Re-implement fixes} (s2.south);

	\caption{Five-step contribution workflow for repository maintenance, from branch creation to merge approval}
	\label{fig:contribution-workflow}

\section{Documentation Updates}

\begin{itemize}
    \item Update API documentation when modifying class signatures
    \item Maintain changelog (CHANGELOG.md) with all modifications
    \item Document algorithm parameter tuning experiences
    \item Keep config file examples synchronized with code
\end{itemize}

\section{Performance Monitoring}

\begin{itemize}
    \item Profile code after major algorithm changes
    \item Compare frame processing times on reference trajectories
    \item Monitor memory usage over long runs (check for leaks)
    \item Track pose optimization convergence times
\end{itemize}

\section{Dependency Management}

\begin{itemize}
    \item Keep OpenCV, g2o, and ROS packages updated
    \item Document minimum version requirements
    \item Test compatibility after system updates
\end{itemize}
